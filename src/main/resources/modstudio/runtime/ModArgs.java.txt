package {{package}};

import com.mojang.brigadier.arguments.ArgumentType;
import com.mojang.brigadier.arguments.BoolArgumentType;
import com.mojang.brigadier.arguments.DoubleArgumentType;
import com.mojang.brigadier.arguments.IntegerArgumentType;
import com.mojang.brigadier.arguments.StringArgumentType;
import com.mojang.brigadier.context.CommandContext;
import com.mojang.brigadier.context.ParsedCommandNode;
import com.mojang.brigadier.tree.ArgumentCommandNode;
import net.minecraft.commands.CommandSourceStack;
import net.minecraft.commands.Commands;
import net.minecraft.network.chat.Component;

import java.util.ArrayList;
import java.util.List;

/**
 * Runs command branches: typed arguments are copied to the data storage {{modid}}:args (path a), whole numbers also
 * to the scores $name {{modid}}.args, then the branch function runs. Generated by ModStudio.
 */
public final class ModArgs {
    private ModArgs() {
    }

    private static final String MODID = "{{modid}}";
    private static final boolean MACROS = {{macros}};

    public static int run(CommandContext<CommandSourceStack> c, String function, int permission) {
        CommandSourceStack source = c.getSource();
        if (!source.hasPermission(permission)) {
            source.sendFailure(Component.literal("You do not have permission to use this command"));
            return 0;
        }
        StringBuilder nbt = new StringBuilder("{");
        List<String> scores = new ArrayList<>();
        for (ParsedCommandNode<CommandSourceStack> node : c.getNodes()) {
            if (!(node.getNode() instanceof ArgumentCommandNode<?, ?> arg)) continue;
            String name = arg.getName();
            String raw = node.getRange().get(c.getInput());
            ArgumentType<?> type = arg.getType();
            String value;
            if (type instanceof IntegerArgumentType) {
                value = raw;
                scores.add("scoreboard players set $" + name + " " + MODID + ".args " + raw);
            } else if (type instanceof DoubleArgumentType) {
                value = raw + "d";
            } else if (type instanceof BoolArgumentType) {
                value = raw;
            } else if (type instanceof StringArgumentType) {
                value = quote(StringArgumentType.getString(c, name));
            } else {
                value = quote(raw);
            }
            if (nbt.length() > 1) nbt.append(',');
            nbt.append(name).append(':').append(value);
        }
        nbt.append('}');
        CommandSourceStack s = source.withPermission(4).withSuppressedOutput();
        Commands commands = source.getServer().getCommands();
        commands.performPrefixedCommand(s, "data modify storage " + MODID + ":args a set value " + nbt);
        for (String x : scores) commands.performPrefixedCommand(s, x);
        commands.performPrefixedCommand(s, "function " + MODID + ":" + function + (MACROS ? " with storage " + MODID + ":args a" : ""));
        return 1;
    }

    /** Checks a choice argument; reports the allowed values when it does not match. */
    public static boolean choice(CommandContext<CommandSourceStack> c, String name, String... values) {
        String v = StringArgumentType.getString(c, name);
        for (String x : values) if (x.equals(v)) return true;
        c.getSource().sendFailure(Component.literal(name + " must be one of: " + String.join(", ", values)));
        return false;
    }

    private static String quote(String s) {
        return "\"" + s.replace("\\", "\\\\").replace("\"", "\\\"") + "\"";
    }
}
