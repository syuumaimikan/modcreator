/**
 * Web requests, secrets and real-time schedules. Requests run on a background thread; results are written to the
 * data storage {{modid}}:api and the matching function runs on the server thread. Generated by ModStudio.
 */
public final class WebBridge {
    private WebBridge() {
    }

    /** Platform glue supplied by the mod or plugin. */
    public interface Host {
        void runOnMain(Runnable task);

        void command(String command);

        Path secretsFile();

        void log(String message);
    }

    public record Extract(String path, String key, boolean toScore) {
    }

    public record Spec(String name, String method, String url, List<String> headers, String body, int timeout, int retries,
                       List<Extract> extract, String schedule, int interval, String cron) {
    }

    private record Job(String cron, String function) {
    }

    private static final String MODID = "{{modid}}";
    private static final int BODY_LIMIT = 8192;
    private static final Map<String, Spec> SPECS = new LinkedHashMap<>();
    private static final List<Job> JOBS = new ArrayList<>();
    private static final Pattern PLACEHOLDER = Pattern.compile("\\{(secret|player|uuid|uuid_plain|time)(?::([A-Za-z0-9_.-]+))?\\}");
    private static final Pattern PATH = Pattern.compile("([^.\\[\\]]+)|\\[(\\d+)\\]");
    private static final Properties SECRETS = new Properties();
    private static final Set<String> WARNED = new HashSet<>();

    private static volatile Host host;
    private static HttpClient client;
    private static ScheduledExecutorService timer;
    private static LocalDateTime lastMinute;

    public static void define(Spec spec) {
        SPECS.put(spec.name(), spec);
    }

    /** Runs a function whenever the five-field cron expression matches the local time. */
    public static void cron(String expression, String function) {
        JOBS.add(new Job(expression, function));
    }

    public static Collection<String> names() {
        return SPECS.keySet();
    }

    public static synchronized void start(Host h) {
        stop();
        host = h;
        loadSecrets(h);
        client = HttpClient.newBuilder().connectTimeout(Duration.ofSeconds(10)).followRedirects(HttpClient.Redirect.NORMAL).build();
        timer = Executors.newSingleThreadScheduledExecutor(r -> {
            Thread t = new Thread(r, MODID + "-web");
            t.setDaemon(true);
            return t;
        });
        boolean cron = !JOBS.isEmpty();
        for (Spec s : SPECS.values()) {
            if (s.schedule().equals("interval")) {
                long n = Math.max(5, s.interval());
                timer.scheduleAtFixedRate(() -> request(s.name(), null, ""), n, n, TimeUnit.SECONDS);
            }
            cron |= s.schedule().equals("cron");
        }
        if (cron) {
            lastMinute = LocalDateTime.now().withSecond(0).withNano(0);
            timer.scheduleAtFixedRate(WebBridge::minute, 5, 15, TimeUnit.SECONDS);
        }
    }

    public static synchronized void stop() {
        if (timer != null) timer.shutdownNow();
        timer = null;
    }

    private static void minute() {
        LocalDateTime now = LocalDateTime.now().withSecond(0).withNano(0);
        if (now.equals(lastMinute)) return;
        lastMinute = now;
        Host h = host;
        if (h == null) return;
        for (Job j : JOBS) {
            if (matches(j.cron(), now)) h.runOnMain(() -> h.command("function " + MODID + ":" + j.function()));
        }
        for (Spec s : SPECS.values()) {
            if (s.schedule().equals("cron") && matches(s.cron(), now)) request(s.name(), null, "");
        }
    }

    // ------------------------------------------------------------------ requests

    /** Starts a request; uuid and player are blank when no player triggered it. */
    public static void request(String name, String uuid, String player) {
        Spec s = SPECS.get(name);
        Host h = host;
        if (s == null || h == null || client == null) return;
        Map<String, String> vars = new LinkedHashMap<>();
        vars.put("player", player == null ? "" : player);
        vars.put("uuid", uuid == null ? "" : uuid);
        vars.put("uuid_plain", uuid == null ? "" : uuid.replace("-", ""));
        vars.put("time", Long.toString(System.currentTimeMillis() / 1000L));
        HttpRequest.Builder b;
        try {
            b = HttpRequest.newBuilder(URI.create(fill(s.url(), vars, 'u'))).timeout(Duration.ofSeconds(Math.max(1, s.timeout())));
        } catch (IllegalArgumentException e) {
            finish(s, uuid, -1, "", "invalid url: " + e.getMessage());
            return;
        }
        boolean contentType = false;
        for (String line : s.headers()) {
            int c = line.indexOf(':');
            if (c <= 0) continue;
            String key = line.substring(0, c).trim();
            contentType |= key.equalsIgnoreCase("content-type");
            try {
                b.header(key, fill(line.substring(c + 1).trim(), vars, 'h'));
            } catch (IllegalArgumentException e) {
                h.log("Skipped header " + key + ": " + e.getMessage());
            }
        }
        String rawBody = s.body() == null ? "" : s.body().trim();
        String body = fill(s.body(), vars, rawBody.startsWith("{") || rawBody.startsWith("[") ? 'j' : 'r');
        boolean hasBody = !body.isEmpty() && !s.method().equals("GET") && !s.method().equals("DELETE");
        if (hasBody && !contentType) {
            String t = body.trim();
            b.header("Content-Type", t.startsWith("{") || t.startsWith("[") ? "application/json" : "text/plain; charset=utf-8");
        }
        b.method(s.method(), hasBody ? HttpRequest.BodyPublishers.ofString(body) : HttpRequest.BodyPublishers.noBody());
        send(s, b.build(), uuid, Math.max(0, s.retries()));
    }

    private static void send(Spec s, HttpRequest req, String uuid, int retriesLeft) {
        HttpClient c = client;
        if (c == null) return;
        c.sendAsync(req, HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8)).whenComplete((res, err) -> {
            boolean retry = retriesLeft > 0 && (err != null || res.statusCode() >= 500 || res.statusCode() == 429);
            ScheduledExecutorService t = timer;
            if (retry && t != null) {
                try {
                    t.schedule(() -> send(s, req, uuid, retriesLeft - 1), 2, TimeUnit.SECONDS);
                    return;
                } catch (RejectedExecutionException ignored) {
                }
            }
            if (err != null) finish(s, uuid, -1, "", err.getClass().getSimpleName() + (err.getMessage() == null ? "" : ": " + err.getMessage()));
            else finish(s, uuid, res.statusCode(), res.body() == null ? "" : res.body(), "");
        });
    }

    private static void finish(Spec s, String uuid, int status, String body, String error) {
        Host h = host;
        if (h == null) return;
        boolean player = uuid != null && !uuid.isEmpty();
        JsonElement json = null;
        try {
            if (!body.isBlank()) json = JsonParser.parseString(body);
        } catch (RuntimeException ignored) {
        }
        List<String> cmds = new ArrayList<>();
        StringBuilder nbt = new StringBuilder("{status:").append(status);
        String holder = player ? uuid : "#" + s.name();
        for (Extract x : s.extract()) {
            JsonElement v = json == null ? null : path(json, x.path());
            nbt.append(',').append(key(x.key())).append(':').append(snbt(v));
            if (x.toScore()) cmds.add("scoreboard players set " + holder + " " + x.key() + " " + score(v));
        }
        nbt.append(",body:").append(quote(body.length() > BODY_LIMIT ? body.substring(0, BODY_LIMIT) : body));
        if (!error.isEmpty()) nbt.append(",error:").append(quote(error));
        nbt.append('}');
        cmds.add(0, "data modify storage " + MODID + ":api " + s.name() + " set value " + nbt);
        cmds.add("scoreboard players set #" + s.name() + " " + MODID + ".http " + status);
        String outcome = status < 0 ? "failure" : status >= 500 ? "server_error" : status >= 400 ? "client_error" : "success";
        String fn = "function " + MODID + ":api/" + s.name() + "/" + outcome;
        if (player) {
            cmds.add("execute as " + uuid + " at @s run " + fn);
            cmds.add("execute unless entity " + uuid + " run " + fn);
        } else {
            cmds.add(fn);
        }
        if (!error.isEmpty()) h.log("Request " + s.name() + " failed: " + error);
        h.runOnMain(() -> {
            for (String c : cmds) h.command(c);
        });
    }

    /**
     * Replaces placeholders. Mode 'u' URL-encodes player values (secrets stay as written, so a whole webhook URL
     * can be a secret), 'j' escapes values for a JSON string, 'h' strips line breaks, 'r' inserts values as they are.
     */
    private static String fill(String template, Map<String, String> vars, char mode) {
        if (template == null || template.isEmpty()) return "";
        Matcher m = PLACEHOLDER.matcher(template);
        StringBuilder sb = new StringBuilder();
        while (m.find()) {
            String v;
            boolean secret = m.group(1).equals("secret");
            if (secret) {
                String name = m.group(2) == null ? "" : m.group(2);
                v = SECRETS.getProperty(name, "");
                if (v.isEmpty() && WARNED.add(name)) {
                    Host h = host;
                    if (h != null) h.log("Secret " + name + " is empty; set it in " + h.secretsFile());
                }
            } else {
                v = vars.getOrDefault(m.group(1), "");
            }
            v = switch (mode) {
                case 'u' -> secret ? v : URLEncoder.encode(v, StandardCharsets.UTF_8).replace("+", "%20");
                case 'j' -> jsonEscape(v);
                case 'h' -> v.replace("\r", "").replace("\n", "");
                default -> v;
            };
            m.appendReplacement(sb, Matcher.quoteReplacement(v));
        }
        m.appendTail(sb);
        return sb.toString();
    }

    private static String jsonEscape(String s) {
        StringBuilder sb = new StringBuilder();
        for (char c : s.toCharArray()) {
            switch (c) {
                case '\\' -> sb.append("\\\\");
                case '"' -> sb.append("\\\"");
                case '\n' -> sb.append("\\n");
                case '\r' -> sb.append("\\r");
                case '\t' -> sb.append("\\t");
                default -> {
                    if (c < 0x20) sb.append(String.format("\\u%04x", (int) c));
                    else sb.append(c);
                }
            }
        }
        return sb.toString();
    }

    // ------------------------------------------------------------------ secrets

    private static void loadSecrets(Host h) {
        SECRETS.clear();
        WARNED.clear();
        Path file = h.secretsFile();
        try {
            if (Files.exists(file)) {
                try (Reader r = Files.newBufferedReader(file, StandardCharsets.UTF_8)) {
                    SECRETS.load(r);
                }
                return;
            }
            Set<String> names = new LinkedHashSet<>();
            for (Spec s : SPECS.values()) {
                List<String> parts = new ArrayList<>(s.headers());
                parts.add(s.url());
                parts.add(s.body());
                for (String p : parts) {
                    Matcher m = PLACEHOLDER.matcher(p);
                    while (m.find()) if (m.group(1).equals("secret") && m.group(2) != null) names.add(m.group(2));
                }
            }
            if (names.isEmpty()) return;
            Path dir = file.toAbsolutePath().getParent();
            if (dir != null) Files.createDirectories(dir);
            try (Writer w = Files.newBufferedWriter(file, StandardCharsets.UTF_8)) {
                w.write("# Secrets for " + MODID + ". Keep this file private; it is never part of the mod.\n");
                for (String n : names) w.write(n + "=\n");
            }
            h.log("Created " + file + "; fill in the secret values and run /reload");
        } catch (IOException e) {
            h.log("Could not read secrets: " + e.getMessage());
        }
    }

    // ------------------------------------------------------------------ JSON

    /** Follows a dot path such as player.stats.level or items[0].name. */
    static JsonElement path(JsonElement root, String path) {
        JsonElement cur = root;
        Matcher m = PATH.matcher(path == null ? "" : path.trim());
        while (m.find()) {
            if (cur == null || cur.isJsonNull()) return null;
            if (m.group(1) != null) {
                if (!cur.isJsonObject()) return null;
                cur = cur.getAsJsonObject().get(m.group(1));
            } else {
                if (!cur.isJsonArray()) return null;
                JsonArray a = cur.getAsJsonArray();
                int i = Integer.parseInt(m.group(2));
                cur = i < a.size() ? a.get(i) : null;
            }
        }
        return cur;
    }

    private static String snbt(JsonElement v) {
        if (v == null || v.isJsonNull()) return "\"\"";
        if (v.isJsonPrimitive()) {
            JsonPrimitive p = v.getAsJsonPrimitive();
            if (p.isBoolean()) return p.getAsBoolean() ? "1b" : "0b";
            if (p.isNumber()) {
                double d = p.getAsDouble();
                if (d == Math.rint(d) && Math.abs(d) <= Integer.MAX_VALUE) return Integer.toString((int) d);
                if (d == Math.rint(d) && Math.abs(d) < 9.0E18) return (long) d + "L";
                return d + "d";
            }
            return quote(p.getAsString());
        }
        return quote(v.toString());
    }

    private static int score(JsonElement v) {
        if (v == null || v.isJsonNull()) return 0;
        if (v.isJsonArray()) return v.getAsJsonArray().size();
        if (v.isJsonObject()) return ((JsonObject) v).size();
        JsonPrimitive p = v.getAsJsonPrimitive();
        if (p.isBoolean()) return p.getAsBoolean() ? 1 : 0;
        try {
            double d = p.isNumber() ? p.getAsDouble() : Double.parseDouble(p.getAsString().trim());
            return (int) Math.max(Integer.MIN_VALUE, Math.min(Integer.MAX_VALUE, Math.round(d)));
        } catch (NumberFormatException e) {
            return 0;
        }
    }

    private static String key(String k) {
        return k.matches("[A-Za-z0-9_]+") ? k : quote(k);
    }

    static String quote(String s) {
        StringBuilder sb = new StringBuilder("\"");
        for (char c : s.toCharArray()) {
            if (c == '\\' || c == '"') sb.append('\\').append(c);
            else if (c < 0x20) sb.append(' ');
            else sb.append(c);
        }
        return sb.append('"').toString();
    }

    // ------------------------------------------------------------------ cron

    /** Five fields: minute hour day-of-month month day-of-week (0 or 7 is Sunday). */
    static boolean matches(String expr, LocalDateTime t) {
        try {
            String[] f = expr.trim().toLowerCase(Locale.ROOT).split("\\s+");
            if (f.length != 5) return false;
            int dow = t.getDayOfWeek().getValue() % 7;
            boolean dom = field(f[2], t.getDayOfMonth(), 1, 31);
            boolean dw = field(f[4], dow, 0, 7) || (dow == 0 && field(f[4], 7, 0, 7));
            boolean day = f[2].equals("*") || f[4].equals("*") ? dom && dw : dom || dw;
            return day && field(f[0], t.getMinute(), 0, 59) && field(f[1], t.getHour(), 0, 23) && field(f[3], t.getMonthValue(), 1, 12);
        } catch (RuntimeException e) {
            return false;
        }
    }

    private static boolean field(String f, int v, int min, int max) {
        for (String part : f.split(",")) {
            int step = 1;
            String range = part;
            int slash = part.indexOf('/');
            if (slash >= 0) {
                step = Math.max(1, Integer.parseInt(part.substring(slash + 1)));
                range = part.substring(0, slash);
            }
            int lo, hi;
            if (range.equals("*")) {
                lo = min;
                hi = max;
            } else if (range.contains("-")) {
                String[] ab = range.split("-");
                lo = Integer.parseInt(ab[0]);
                hi = Integer.parseInt(ab[1]);
            } else {
                lo = Integer.parseInt(range);
                hi = slash >= 0 ? max : lo;
            }
            if (v >= lo && v <= hi && (v - lo) % step == 0) return true;
        }
        return false;
    }
